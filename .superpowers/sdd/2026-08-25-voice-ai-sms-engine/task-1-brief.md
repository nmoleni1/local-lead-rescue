```sql
-- Calls table: stores Twilio call records and transcripts
CREATE TABLE IF NOT EXISTS calls (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    call_sid TEXT UNIQUE,
    from_number TEXT NOT NULL,
    to_number TEXT NOT NULL,
    duration INTEGER DEFAULT 0,
    transcript TEXT,
    recording_url TEXT,
    status TEXT DEFAULT 'completed',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Messages table: stores threaded 2-way SMS conversations
CREATE TABLE IF NOT EXISTS messages (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    lead_id BIGINT REFERENCES leads(id) ON DELETE CASCADE,
    from_number TEXT NOT NULL,
    to_number TEXT NOT NULL,
    direction TEXT CHECK (direction IN ('inbound', 'outbound')),
    body TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Add new columns to leads table
ALTER TABLE leads ADD COLUMN IF NOT EXISTS address TEXT;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS preferred_window TEXT DEFAULT 'Flexible';
ALTER TABLE leads ADD COLUMN IF NOT EXISTS call_sid TEXT;

-- RLS policies for calls
ALTER TABLE calls ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all access to calls" ON calls FOR ALL USING (true) WITH CHECK (true);

-- RLS policies for messages
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all access to messages" ON messages FOR ALL USING (true) WITH CHECK (true);
```

```python
        "call_sid": call_sid,

        "direction": direction,
        "body": body
    }
    res = supabase_request("messages", method="POST", data=payload)

        return res[0].get("id")
    return None

def get_messages_for_lead(lead_id):
    """Fetch all messages for a given lead ordered by time ascending."""
    res = supabase_request(f"messages?lead_id=eq.{lead_id}&order=created_at.asc&select=*")
    if res and isinstance(res, list):
        return res
    return []

def get_all_leads():
    """Fetch all leads from Supabase ordered newest first, including message thread count."""
    res = supabase_request("leads?select=*&order=created_at.desc")
    if res and isinstance(res, list):
        return res
```
